Script de la base de datos y la tabla solicitudes. 

-- Crear la base de datos
CREATE DATABASE IF NOT EXISTS reeduca_db CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;

-- Usar la base de datos
USE reeduca_db;

-- Crear tabla "solicitudes"
CREATE TABLE IF NOT EXISTS solicitudes (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL,
  documento VARCHAR(50) NOT NULL,
  telefono VARCHAR(20),
  correo VARCHAR(100),
  departamento VARCHAR(50) NOT NULL,
  nivel_estudio VARCHAR(50) NOT NULL,
  desplazado VARCHAR(10) NOT NULL,
  mensaje TEXT,
  fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
